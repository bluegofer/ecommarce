// Branch + Supplier + Purchase + GRN + Supplier Payment + Supplier Due

enum BranchStatus {
  ACTIVE
  INACTIVE
}

enum SupplierStatus {
  ACTIVE
  INACTIVE
  BLACKLISTED
}

enum PurchaseRequisitionStatus {
  DRAFT
  SUBMITTED
  APPROVED
  REJECTED
  CONVERTED
}

enum PurchaseOrderStatus {
  DRAFT
  SENT
  PARTIAL_RECEIVED
  RECEIVED
  CLOSED
  CANCELLED
}

enum PurchaseInvoiceStatus {
  DRAFT
  POSTED
  PARTIAL_PAID
  PAID
  VOID
}

enum GRNStatus {
  DRAFT
  CONFIRMED
  CANCELLED
}

enum SupplierPaymentMethod {
  CASH
  BANK
  MFS
}

enum DueStatus {
  OPEN
  PARTIAL
  PAID
  OVERDUE
  WRITTEN_OFF
}

model Branch {
  id        String       @id @default(cuid())
  code      String       @unique
  name      String
  nameBn    String?
  address   Json?
  phone     String?
  email     String?
  isDefault Boolean      @default(false)
  status    BranchStatus @default(ACTIVE)
  createdAt DateTime     @default(now())
  updatedAt DateTime     @updatedAt

  purchaseRequisitions PurchaseRequisition[]
  purchaseOrders       PurchaseOrder[]
  grns                 GRN[]

  @@index([status])
  @@map("branches")
}

model Supplier {
  id             String         @id @default(cuid())
  code           String         @unique
  name           String
  contactPerson  String?
  phone          String?
  email          String?
  address        Json?
  taxId          String?
  paymentTerms   String?
  notes          String?
  openingBalance Int            @default(0)
  currentDue     Int            @default(0)
  status         SupplierStatus @default(ACTIVE)
  createdById    String?
  createdAt      DateTime       @default(now())
  updatedAt      DateTime       @updatedAt

  purchaseOrders   PurchaseOrder[]
  purchaseInvoices PurchaseInvoice[]
  payments         SupplierPayment[]
  dues             SupplierDue[]

  @@index([status])
  @@index([name])
  @@map("suppliers")
}

model PurchaseRequisition {
  id            String                    @id @default(cuid())
  reqNumber     String                    @unique
  branchId      String
  branch        Branch                    @relation(fields: [branchId], references: [id])
  status        PurchaseRequisitionStatus @default(DRAFT)
  notes         String?
  requestedById String?
  approvedById  String?
  approvedAt    DateTime?
  createdAt     DateTime                  @default(now())
  updatedAt     DateTime                  @updatedAt

  items          PurchaseRequisitionItem[]
  purchaseOrders PurchaseOrder[]

  @@index([status])
  @@index([branchId])
  @@map("purchase_requisitions")
}

model PurchaseRequisitionItem {
  id            String              @id @default(cuid())
  requisitionId String
  requisition   PurchaseRequisition @relation(fields: [requisitionId], references: [id], onDelete: Cascade)
  variantId     String
  requestedQty  Int
  notes         String?
  createdAt     DateTime            @default(now())

  @@index([requisitionId])
  @@index([variantId])
  @@map("purchase_requisition_items")
}

model PurchaseOrder {
  id            String               @id @default(cuid())
  poNumber      String               @unique
  supplierId    String
  supplier      Supplier             @relation(fields: [supplierId], references: [id])
  branchId      String
  branch        Branch               @relation(fields: [branchId], references: [id])
  requisitionId String?
  requisition   PurchaseRequisition? @relation(fields: [requisitionId], references: [id])
  status        PurchaseOrderStatus  @default(DRAFT)
  expectedDate  DateTime?
  subtotal      Int                  @default(0)
  discount      Int                  @default(0)
  total         Int                  @default(0)
  notes         String?
  createdById   String?
  sentAt        DateTime?
  closedAt      DateTime?
  createdAt     DateTime             @default(now())
  updatedAt     DateTime             @updatedAt

  items    PurchaseOrderItem[]
  invoices PurchaseInvoice[]
  grns     GRN[]

  @@index([status])
  @@index([supplierId])
  @@index([branchId])
  @@map("purchase_orders")
}

model PurchaseOrderItem {
  id            String        @id @default(cuid())
  poId          String
  purchaseOrder PurchaseOrder @relation(fields: [poId], references: [id], onDelete: Cascade)
  variantId     String
  orderedQty    Int
  receivedQty   Int           @default(0)
  unitCost      Int
  lineTotal     Int
  notes         String?
  createdAt     DateTime      @default(now())

  grnItems GRNItem[]

  @@index([poId])
  @@index([variantId])
  @@map("purchase_order_items")
}

model PurchaseInvoice {
  id              String                @id @default(cuid())
  invoiceNumber   String
  supplierId      String
  supplier        Supplier              @relation(fields: [supplierId], references: [id])
  purchaseOrderId String?
  purchaseOrder   PurchaseOrder?        @relation(fields: [purchaseOrderId], references: [id])
  invoiceDate     DateTime
  dueDate         DateTime?
  subtotal        Int                   @default(0)
  discount        Int                   @default(0)
  tax             Int                   @default(0)
  total           Int                   @default(0)
  paidAmount      Int                   @default(0)
  status          PurchaseInvoiceStatus @default(DRAFT)
  notes           String?
  postedAt        DateTime?
  createdAt       DateTime              @default(now())
  updatedAt       DateTime              @updatedAt

  payments SupplierPayment[]
  dues     SupplierDue[]

  @@unique([supplierId, invoiceNumber])
  @@index([status])
  @@index([supplierId])
  @@map("purchase_invoices")
}

model GRN {
  id              String    @id @default(cuid())
  grnNumber       String    @unique
  purchaseOrderId String
  purchaseOrder   PurchaseOrder @relation(fields: [purchaseOrderId], references: [id])
  branchId        String
  branch          Branch    @relation(fields: [branchId], references: [id])
  status          GRNStatus @default(DRAFT)
  receivedById    String?
  receivedAt      DateTime?
  notes           String?
  createdAt       DateTime  @default(now())
  updatedAt       DateTime  @updatedAt

  items GRNItem[]

  @@index([status])
  @@index([purchaseOrderId])
  @@map("grns")
}

model GRNItem {
  id          String             @id @default(cuid())
  grnId       String
  grn         GRN                @relation(fields: [grnId], references: [id], onDelete: Cascade)
  poItemId    String?
  poItem      PurchaseOrderItem? @relation(fields: [poItemId], references: [id])
  variantId   String
  receivedQty Int
  unitCost    Int
  lineTotal   Int
  notes       String?
  createdAt   DateTime           @default(now())

  @@index([grnId])
  @@index([variantId])
  @@map("grn_items")
}

model SupplierPayment {
  id                String                @id @default(cuid())
  paymentNumber     String                @unique
  supplierId        String
  supplier          Supplier              @relation(fields: [supplierId], references: [id])
  purchaseInvoiceId String?
  purchaseInvoice   PurchaseInvoice?      @relation(fields: [purchaseInvoiceId], references: [id])
  amount            Int
  method            SupplierPaymentMethod
  ledgerAccountId   String?
  journalEntryId    String?
  reference         String?
  paidAt            DateTime
  notes             String?
  createdById       String?
  createdAt         DateTime              @default(now())

  @@index([supplierId])
  @@index([purchaseInvoiceId])
  @@index([paidAt])
  @@map("supplier_payments")
}

model SupplierDue {
  id                String           @id @default(cuid())
  supplierId        String
  supplier          Supplier         @relation(fields: [supplierId], references: [id])
  purchaseInvoiceId String?
  purchaseInvoice   PurchaseInvoice? @relation(fields: [purchaseInvoiceId], references: [id])
  amount            Int
  paidAmount        Int              @default(0)
  balance           Int
  dueDate           DateTime?
  status            DueStatus        @default(OPEN)
  notes             String?
  createdAt         DateTime         @default(now())
  updatedAt         DateTime         @updatedAt

  @@index([supplierId])
  @@index([status])
  @@index([dueDate])
  @@map("supplier_dues")
}
